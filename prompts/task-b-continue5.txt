def _mask_config(key, value):
    upper = key.upper()
    if any(sub in upper for sub in ("SECRET", "PASSWORD", "TOKEN", "KEY")):
        return "***"
    try:
        json.dumps(value)
        return value
    except (TypeError, ValueError):
        return str(value)
